cmake_minimum_required(VERSION 3.16)
project(pocketds_powerdevil_battery_health LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core Qml)
if(NOT EXISTS "${BATTERY_ITEM_SOURCE}")
    message(FATAL_ERROR "Set BATTERY_ITEM_SOURCE to the patched upstream BatteryItem.qml")
endif()
add_executable(battery-health battery-health.cpp)
target_link_libraries(battery-health PRIVATE Qt6::Core Qt6::Qml)
enable_testing()
add_test(NAME battery-health COMMAND battery-health "${BATTERY_ITEM_SOURCE}")
set_tests_properties(battery-health PROPERTIES TIMEOUT 10 ENVIRONMENT "QML_DISABLE_DISK_CACHE=1")
