cmake_minimum_required(VERSION 3.16)
project(pocketds_powerdevil_resume_gate LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core DBus)
find_program(DBUS_RUN_SESSION dbus-run-session REQUIRED)
add_executable(resume-gate resume-gate.cpp)
target_include_directories(resume-gate PRIVATE ..)
target_link_libraries(resume-gate PRIVATE Qt6::Core Qt6::DBus)
enable_testing()
add_test(NAME resume-gate COMMAND "${DBUS_RUN_SESSION}" -- $<TARGET_FILE:resume-gate>)
set_tests_properties(resume-gate PROPERTIES TIMEOUT 25)
